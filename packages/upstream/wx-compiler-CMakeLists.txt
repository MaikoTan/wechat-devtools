cmake_minimum_required(VERSION 3.5)
project(wx_complier VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -D_REENTRANT")

if(NOT DEFINED TARGET_ARCH)
    set(TARGET_ARCH x64)
endif()

# node-addon-api 头文件目录（由 Nix 提供）
set(NODE_ADDON_API_DIR "@NODE_ADDON_API_DIR@" CACHE PATH "node-addon-api include dir")
set(NODE_API_HEADERS_DIR "@NODE_API_HEADERS_DIR@" CACHE PATH "node_api.h dir")

# nixpkgs 的 gcc 13 头文件在 <iostream> 里声明了 _ZSt21ios_base_library_initv，
# 它只在 gcc 15 的 libstdc++ 里提供（GLIBCXX_3.4.32）。
# Electron 36.6.0 内置的 libstdc++ 是 gcc 13，最高只到 GLIBCXX_3.4.29，
# 不处理这个符号会让模块在 Electron 进程里加载失败。
# wcc/wcsc 只用 iostream 打日志，给个空实现即可。
add_library(iostream_stub STATIC @IOSTREAM_STUB@)

# 可执行文件：wcc / wcsc
add_executable(wcc

    src/include/wxml.h

    src/night/night.cpp

    src/night/ns_compile_js.cpp
    src/night/ns_node.cpp
    src/night/ns_sourcemap.cpp
    src/night/ns_stream.cpp
    src/night/ns_token.cpp
    src/night/ns_god/ns_god.cpp
    src/night/str/str.cpp
    src/wxml/rpx.cpp
    src/wxml/expr_lib/base.cpp

    src/utils/string_utils.cpp
    src/utils/file.cpp
    )
target_link_libraries(wcc PRIVATE iostream_stub)


add_executable(wcsc

    src/wcsc/usage.cpp
    src/wxa/wxa.cpp

    src/utils/bittest.cpp
